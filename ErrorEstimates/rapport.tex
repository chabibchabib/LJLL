% ---- macros ----------------------------------------------------------
% #1 = dossier de la source, #2 = nom affiche
\newcommand{\errframe}[2]{%
  \begin{frame}{#2 --- Relative errors}
    \centering
    \includegraphics[width=.95\textwidth,height=.75\textheight,keepaspectratio]{#1/err.pdf}
  \end{frame}}

% #1 = source, #2 = nom, #3,#4 = deux valeurs de k
\newcommand{\meshframe}[4]{%
  \begin{frame}{#2 --- Meshes of Levels: 0 and 4 ( Initial and last ones)}
    \centering
    \includegraphics[height=.40\textheight,keepaspectratio]{#1/mesh_k#3.pdf}\\[2mm]
    \includegraphics[height=.40\textheight,keepaspectratio]{#1/mesh_k#4.pdf}
  \end{frame}}

\newcommand{\solframe}[4]{%
  \begin{frame}{#2 --- Solutions : Reference, Level 0, Level 4}
    \centering
    \includegraphics[height=.40\textheight,keepaspectratio]{#1/sol_k#3.pdf}\\[2mm]
    \includegraphics[height=.40\textheight,keepaspectratio]{#1/sol_k#4.pdf}
  \end{frame}}

% regroupe tout ce qui concerne une source (hors solutions)
